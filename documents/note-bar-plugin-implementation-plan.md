# Note Bar Plugin — Obsidian 浮动格式工具栏插件实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 开发一个 Obsidian 插件，在 Markdown 编辑器中选中文本时弹出类似飞书云文档风格的浮动格式工具栏，支持文本样式、对齐缩进和格式按钮组功能。

**Architecture:** 基于 Obsidian Plugin API + 原生 DOM 构建轻量化浮动工具栏。通过 `Editor` 抽象层操作文本，使用 CSS `backdrop-filter` 实现毛玻璃效果，监听编辑器选区变化事件触发显示/隐藏。插件无外部 UI 框架依赖，保持极小体积。

**Tech Stack:** TypeScript, Obsidian API (Plugin, Editor), esbuild, CSS (backdrop-filter frosted glass)

```typescript
import { Editor } from "obsidian";

export interface ToolbarPosition {
  top: number;
  left: number;
}

/**
 * 获取编辑器选区在视口中的位置（像素坐标）
 * 使用 Selection API 获取 Range 的 bounding rect
 */
export function getSelectionPosition(editor: Editor): ToolbarPosition | null {
  const activeView = document.querySelector(".cm-editor .cm-content");
  if (!activeView) return null;

  const selection = activeView.ownerDocument.getSelection();
  if (!selection || selection.isCollapsed || !selection.rangeCount) return null;

  const range = selection.getRangeAt(0);
  const rect = range.getBoundingClientRect();

  if (!rect || (rect.width === 0 && rect.height === 0)) return null;

  return {
    top: rect.top,
    left: rect.left + rect.width / 2,
  };
}

/**
 * 计算工具栏的最终位置，确保不超出视口边界
 */
export function calculateToolbarPosition(
  selectionPos: ToolbarPosition,
  toolbarWidth: number,
  toolbarHeight: number,
  gap: number = 8
): ToolbarPosition {
  const viewportWidth = window.innerWidth;
  const viewportHeight = window.innerHeight;

  // 水平居中于选区上方
  let left = selectionPos.left - toolbarWidth / 2;
  // 垂直位于选区上方
  let top = selectionPos.top - toolbarHeight - gap;

  // 防止超出左侧
  if (left < 12) left = 12;
  // 防止超出右侧
  if (left + toolbarWidth + 12 > viewportWidth) {
    left = viewportWidth - toolbarWidth - 12;
  }

  // 如果上方空间不足，显示在选区下方
  if (top < 12) {
    top = selectionPos.top + gap;
  }

  // 防止超出底部
  if (top + toolbarHeight + 12 > viewportHeight) {
    top = viewportHeight - toolbarHeight - 12;
  }

  return { top, left };
}

/**
 * 检查工具栏是否应该隐藏（选区消失或折叠）
 */
export function shouldHideToolbar(editor: Editor): boolean {
  const selection = editor.getSelection();
  return !selection || selection.length === 0;
}
```

```typescript
import { Editor } from "obsidian";
import { HighlightColorKey, HIGHLIGHT_COLORS } from "../constants";

export type TextStyle =
  | "paragraph"
  | "heading1"
  | "heading2"
  | "heading3"
  | "heading4"
  | "heading5"
  | "heading6"
  | "ordered-list"
  | "unordered-list"
  | "task"
  | "code-block"
  | "blockquote"
  | "callout";

export type AlignIndent = "align-left" | "align-center" | "align-right" | "indent-increase" | "indent-decrease";

export type FormatType = "bold" | "strikethrough" | "italic" | "underline" | "link";

/**
 * 应用文本样式（标题、列表、引用等）
 */
export function applyTextStyle(editor: Editor, style: TextStyle): void {
  const cursor = editor.getCursor();
  const line = editor.getLine(cursor.line);
  const currentPrefix = detectLinePrefix(line);

  let newLine: string;

  switch (style) {
    case "paragraph":
      newLine = stripPrefix(line);
      break;
    case "heading1":
      newLine = `# ${stripPrefix(line)}`;
      break;
    case "heading2":
      newLine = `## ${stripPrefix(line)}`;
      break;
    case "heading3":
      newLine = `### ${stripPrefix(line)}`;
      break;
    case "heading4":
      newLine = `#### ${stripPrefix(line)}`;
      break;
    case "heading5":
      newLine = `##### ${stripPrefix(line)}`;
      break;
    case "heading6":
      newLine = `###### ${stripPrefix(line)}`;
      break;
    case "ordered-list":
      newLine = `1. ${stripPrefix(line)}`;
      break;
    case "unordered-list":
      newLine = `- ${stripPrefix(line)}`;
      break;
    case "task":
      newLine = `- [ ] ${stripPrefix(line)}`;
      break;
    case "code-block":
      newLine = "```\n" + line + "\n```";
      editor.setLine(cursor.line, newLine);
      return;
    case "blockquote":
      newLine = `> ${stripPrefix(line)}`;
      break;
    case "callout":
      newLine = `> [!note]\n> ${stripPrefix(line)}`;
      editor.setLine(cursor.line, newLine);
      return;
  }

  editor.setLine(cursor.line, newLine);
}

/**
 * 应用对齐与缩进
 */
export function applyAlignIndent(editor: Editor, action: AlignIndent): void {
  const cursor = editor.getCursor();
  const line = editor.getLine(cursor.line);

  switch (action) {
    case "align-left":
      editor.setLine(cursor.line, `<div style="text-align:left">${line}</div>`);
      break;
    case "align-center":
      editor.setLine(cursor.line, `<div style="text-align:center">${line}</div>`);
      break;
    case "align-right":
      editor.setLine(cursor.line, `<div style="text-align:right">${line}</div>`);
      break;
    case "indent-increase":
      editor.setLine(cursor.line, `\t${line}`);
      break;
    case "indent-decrease":
      if (line.startsWith("\t")) {
        editor.setLine(cursor.line, line.slice(1));
      }
      break;
  }
}

/**
 * 应用内联格式（加粗/斜体等）
 */
export function applyInlineFormat(editor: Editor, type: FormatType): void {
  const selection = editor.getSelection();
  if (!selection) return;

  let wrapped: string;

  switch (type) {
    case "bold":
      wrapped = `**${selection}**`;
      break;
    case "italic":
      wrapped = `*${selection}*`;
      break;
    case "strikethrough":
      wrapped = `~~${selection}~~`;
      break;
    case "underline":
      wrapped = `<u>${selection}</u>`;
      break;
    case "link":
      wrapped = `[${selection}](url)`;
      break;
  }

  editor.replaceSelection(wrapped);
}

/**
 * 应用高亮颜色
 */
export function applyHighlight(editor: Editor, colorKey: HighlightColorKey): void {
  const selection = editor.getSelection();
  if (!selection) return;

  const color = HIGHLIGHT_COLORS[colorKey].value;
  const wrapped = `<mark style="background:${color}">${selection}</mark>`;

  editor.replaceSelection(wrapped);
}

function detectLinePrefix(line: string): string {
  const prefixMatch = line.match(
    /^(\s*)(#{1,6}\s|>\s|-\s\[\s\]\s|-\s|\d+\.\s|```)/
  );
  return prefixMatch ? prefixMatch[0] : "";
}

function stripPrefix(line: string): string {
  return line.replace(
    /^(\s*)(#{1,6}\s|>\s|-\s\[\s\]\s|-\s|\d+\.\s|```\s*)/,
    "$1"
  ).trim();
}
```

import { Editor, Plugin } from "obsidian";
import {
  TOOLBAR_CLASS,
  TOOLBAR_VISIBLE_CLASS,
  TOOLBAR_THEME_LIGHT,
  TOOLBAR_THEME_DARK,
} from "../constants";
import { getSelectionPosition, calculateToolbarPosition, shouldHideToolbar } from "../utils/position-calc";
import { createStyleDropdown } from "./components/StyleDropdown";
import { createAlignDropdown } from "./components/AlignDropdown";
import { createFormatButtons } from "./components/FormatButtons";

export class ToolbarManager {
  private toolbarEl: HTMLElement;
  private plugin: Plugin;
  private isVisible = false;
  private debounceTimer: number | null = null;
  private currentEditor: Editor | null = null;
  private onDismiss: () => void;

  constructor(plugin: Plugin) {
    this.plugin = plugin;
    this.onDismiss = () => this.hide();
    this.toolbarEl = this.createToolbarElement();
  }

  private createToolbarElement(): HTMLElement {
    const el = document.createElement("div");
    el.className = `${TOOLBAR_CLASS} ${TOOLBAR_THEME_LIGHT}`;
    el.style.pointerEvents = "none";

    // 阻止工具栏内部点击冒泡到空白区域
    el.addEventListener("mousedown", (e) => e.stopPropagation());
    el.addEventListener("click", (e) => e.stopPropagation());

    // 构建子组件
    const styleDropdown = createStyleDropdown(this.onDismiss);
    const alignDropdown = createAlignDropdown(this.onDismiss);
    const formatButtons = createFormatButtons(this.onDismiss);

    // 分割线
    const divider = (): HTMLElement => {
      const d = document.createElement("div");
      d.className = "note-bar-divider";
      return d;
    };

    el.appendChild(styleDropdown);
    el.appendChild(divider());
    el.appendChild(alignDropdown);
    el.appendChild(divider());
    el.appendChild(formatButtons);

    document.body.appendChild(el);
    return el;
  }

  /**
   * 响应选区变化
   */
  onSelectionChange(editor: Editor): void {
    this.currentEditor = editor;

    // 防抖处理（<100ms）
    if (this.debounceTimer !== null) {
      window.clearTimeout(this.debounceTimer);
    }

    this.debounceTimer = window.setTimeout(() => {
      this.updateToolbar(editor);
    }, 50);
  }

  private updateToolbar(editor: Editor): void {
    const shouldHide = shouldHideToolbar(editor);

    if (shouldHide) {
      this.hide();
      return;
    }

    const selectionPos = getSelectionPosition(editor);
    if (!selectionPos) {
      this.hide();
      return;
    }

    // 更新主题
    this.updateTheme();

    // 显示工具栏并定位
    this.show(editor, selectionPos);
  }

  private show(editor: Editor, selectionPos: ReturnType<typeof getSelectionPosition>): void {
    if (!selectionPos) return;

    // 先显示以获取实际尺寸
    if (!this.isVisible) {
      this.toolbarEl.classList.add(TOOLBAR_VISIBLE_CLASS);
      this.isVisible = true;
    }

    // 强制回流以获取准确的 offset 尺寸
    const toolbarWidth = this.toolbarEl.offsetWidth || 200;
    const toolbarHeight = this.toolbarEl.offsetHeight || 40;

    const pos = calculateToolbarPosition(selectionPos, toolbarWidth, toolbarHeight);

    this.toolbarEl.style.left = `${pos.left}px`;
    this.toolbarEl.style.top = `${pos.top}px`;
    this.toolbarEl.style.pointerEvents = "auto";
  }

  private hide(): void {
    if (!this.isVisible) return;

    this.toolbarEl.classList.remove(TOOLBAR_VISIBLE_CLASS);
    this.toolbarEl.style.pointerEvents = "none";
    this.isVisible = false;
    this.currentEditor = null;
  }

  private updateTheme(): void {
    const isDark = document.body.classList.contains("theme-dark");
    this.toolbarEl.classList.toggle(TOOLBAR_THEME_LIGHT, !isDark);
    this.toolbarEl.classList.toggle(TOOLBAR_THEME_DARK, isDark);
  }

  /**
   * 全局点击处理：点击空白区域时消失
   */
  onGlobalClick(e: MouseEvent): void {
    if (!this.isVisible) return;

    const target = e.target as HTMLElement;
    // 如果点击目标不在工具栏内部，则隐藏
    if (!this.toolbarEl.contains(target)) {
      this.hide();
    }
  }

  /**
   * ESC 键处理：按下 Esc 时消失
   */
  onKeyDown(e: KeyboardEvent): void {
    if (e.key === "Escape" && this.isVisible) {
      this.hide();
    }
  }

  /**
   * 清理资源
   */
  destroy(): void {
    this.hide();
    this.toolbarEl.remove();
    if (this.debounceTimer !== null) {
      window.clearTimeout(this.debounceTimer);
    }
  }
}
```

          // 展开子菜单
          if (subMenu) {
            subMenu.remove();
            subMenu = null;
            return;
          }
          subMenu = document.createElement("div");
          subMenu.className = "note-bar-submenu";

          const subItems: { label: string; style: TextStyle }[] = [
            { label: "四级标题", style: "heading4" },
            { label: "五级标题", style: "heading5" },
            { label: "六级标题", style: "heading6" },
          ];

          subItems.forEach((sub) => {
            const subEl = document.createElement("div");
            subEl.className = "note-bar-dropdown-item";
            subEl.textContent = sub.label;
            subEl.addEventListener("click", (se) => {
              se.stopPropagation();
              applyTextStyleFromEditor(sub.style);
              closePanel();
              onAction();
            });
            subMenu!.appendChild(subEl);
          });

          // 定位子菜单
          const rect = el.getBoundingClientRect();
          subMenu.style.left = `${rect.right + 4}px`;
          subMenu.style.top = `${rect.top}px`;

          // 使用 requestAnimationFrame 触发动画
          requestAnimationFrame(() => {
            subMenu?.classList.add("note-bar-submenu--open");
          });
        });
      } else {
        el.textContent = item.label;
        el.addEventListener("click", () => {
          applyTextStyleFromEditor(item.style);
          closePanel();
          onAction();
        });
      }

      panel!.appendChild(el);
    });

    // 定位面板
    const rect = trigger.getBoundingClientRect();
    panel.style.left = `${rect.left}px`;
    panel.style.top = `${rect.bottom + 4}px`;
    document.body.appendChild(panel);

    isOpen = true;
    requestAnimationFrame(() => {
      panel?.classList.add(DROPDOWN_OPEN_CLASS);
    });
  };

  const closePanel = () => {
    if (panel) {
      panel.classList.remove(DROPDOWN_OPEN_CLASS);
      setTimeout(() => {
        if (panel) {
          panel.remove();
          panel = null;
        }
      }, 120);
    }
    // 清理子菜单
    document.querySelectorAll(".note-bar-submenu").forEach((el) => el.remove());
    isOpen = false;
  };

  const applyTextStyleFromEditor = (style: TextStyle) => {
    const view = document.querySelector(".cm-editor");
    // 通过 Obsidian API 获取编辑器实例
    // 实际运行时由 ToolbarManager 注入
    const editor = (window as any).__noteBarEditor as Editor;
    if (editor) {
      applyTextStyle(editor, style);
    }
  };

  trigger.addEventListener("click", (e) => {
    e.stopPropagation();
    if (isOpen) closePanel();
    else openPanel();
  });

  // 点击外部关闭面板
  const onDocClick = (e: MouseEvent) => {
    if (isOpen && panel && !panel.contains(e.target as Node) && e.target !== trigger) {
      closePanel();
    }
  };
  document.addEventListener("mousedown", onDocClick);

  // 清理
  const originalRemove = container.remove.bind(container);
  container.remove = () => {
    document.removeEventListener("mousedown", onDocClick);
    closePanel();
    originalRemove();
  } as any;

  return container;
}

import { Editor } from "obsidian";
import { HIGHLIGHT_COLORS, HighlightColorKey } from "../../constants";

export function createHighlightPicker(
  anchor: HTMLElement,

    ));
  }

  label.textContent = "高亮颜色";
  picker.appendChild(label);

    const swatch = document.createElement("div");
    swatch.className = "note-bar-color-swatch";
    swatch.style.background = color.value;
    swatch.title = color.name;

    swatch.addEventListener("click", (e) => {

      // 更新选中状态
      grid.querySelectorAll(".note-bar-color-swatch").forEach((s) => {
        s.classList.remove("note-bar-color-swatch--selected");
      });
      swatch.classList.add("note-bar-color-swatch--selected");

      // 应用高亮
      const editor = (window as any).__noteBarEditor as Editor;
      if (editor) {
        applyHighlight(editor, key);
      }

      onColorChange(key);

      // 关闭选择器
      picker.classList.remove("note-bar-color-picker--open");
      setTimeout(() => picker.remove(), 120);
    });

    grid.appendChild(swatch);
  });

  picker.appendChild(grid);

  // 定位
  const rect = anchor.getBoundingClientRect();
  picker.style.left = `${rect.left}px`;
  picker.style.top = `${rect.bottom + 4}px`;
  document.body.appendChild(picker);

  // 动画
  requestAnimationFrame(() => {
    picker.classList.add("note-bar-color-picker--open");
  });

  // 点击外部关闭
  const onDocClick = (e: MouseEvent) => {
    if (!picker.contains(e.target as Node) && e.target !== anchor) {
      picker.classList.remove("note-bar-color-picker--open");
      setTimeout(() => picker.remove(), 120);
      document.removeEventListener("mousedown", onDocClick);
    }
  };
  setTimeout(() => document.addEventListener("mousedown", onDocClick), 0);
}

```typescript
import { Plugin, MarkdownView } from "obsidian";
import { ToolbarManager } from "./toolbar/ToolbarManager";

export default class NoteBarPlugin extends Plugin {
  private toolbarManager: ToolbarManager | null = null;
  private boundSelectionChange: (() => void) | null = null;
  private boundGlobalClick: ((e: MouseEvent) => void) | null = null;
  private boundKeyDown: ((e: KeyboardEvent) => void) | null = null;

  async onload() {
    console.log("Note Bar plugin loaded");

    this.toolbarManager = new ToolbarManager(this);

    // 绑定事件
    this.boundSelectionChange = this.handleSelectionChange.bind(this);
    this.boundGlobalClick = (e: MouseEvent) => this.toolbarManager?.onGlobalClick(e);
    this.boundKeyDown = (e: KeyboardEvent) => this.toolbarManager?.onKeyDown(e);

    // 监听编辑器选区变化
    this.registerSelectionChangeListener();

    // 全局点击和键盘事件
    document.addEventListener("mousedown", this.boundGlobalClick);
    document.addEventListener("keydown", this.boundKeyDown);

    // 清理
    this.registerOnunload();
  }

  private registerSelectionChangeListener(): void {
    // 通过 MutationObserver 或定时检查视图变化来监听选区
    // 使用更可靠的方式：监听 activeEditorState 变化
    this.registerInterval(
      window.setInterval(() => {
        const editor = this.getActiveEditor();
        if (editor) {
          // 注入 editor 到全局供组件使用
          (window as any).__noteBarEditor = editor;
          this.toolbarManager?.onSelectionChange(editor);
        }
      }, 80) // 80ms 轮询，低于 100ms 要求
    );
  }

  private getActiveEditor() {
    const view = this.app.workspace.getActiveViewOfType(MarkdownView);
    if (view) {
      return view.editor;
    }
    return null;
  }

  private handleSelectionChange(): void {
    const editor = this.getActiveEditor();
    if (editor) {
      (window as any).__noteBarEditor = editor;
      this.toolbarManager?.onSelectionChange(editor);
    }
  }

  private registerOnunload(): void {
    this.register(() => {
      this.onunload();
    });
  }

  async onunload() {
    console.log("Note Bar plugin unloaded");

    if (this.boundGlobalClick) {
      document.removeEventListener("mousedown", this.boundGlobalClick);
    }
    if (this.boundKeyDown) {
      document.removeEventListener("keydown", this.boundKeyDown);
    }

    if (this.toolbarManager) {
      this.toolbarManager.destroy();
      this.toolbarManager = null;
    }

    (window as any).__noteBarEditor = undefined;
  }
}

## 架构设计决策记录

| 决策 | 选择 | 理由 |
|------|------|------|
| UI 框架 | 无（纯 DOM） | 避免引入外部依赖，保持插件体积小、加载快 |
| 文本操作 | Obsidian `Editor` API | 官方推荐，同时兼容 CM5/CM6 |
| 选区检测 | 80ms 轮询 + `selectionchange` | 满足 <100ms 延迟要求，实现简单可靠 |
| 定位计算 | `getBoundingClientRect` | 原生 API，精确获取选区位置 |
| 毛玻璃 | `backdrop-filter: blur(20px)` | 标准 CSS 效果，macOS 原生质感 |
| 主题适配 | CSS class toggle (`theme-dark`/`theme-light`) | 利用 Obsidian 内置主题变量 |
| 消失逻辑 | `mousedown` + `stopPropagation` | 精确控制，防止误触发 |
